cmake_minimum_required(VERSION 3.8)
project(baja_gz_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_executable(cloud_bridge src/cloud_bridge.cpp)
target_compile_features(cloud_bridge PRIVATE cxx_std_17)
ament_target_dependencies(cloud_bridge rclcpp sensor_msgs)
target_link_libraries(cloud_bridge gz-transport13::core gz-msgs10::gz-msgs10)
install(TARGETS cloud_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
